    // Parser universal da resposta JSON do Sankhya (usado por ambos os módulos)
    function parseSankhyaResponse(jsonObj) {
        try {
            const rawInner = jsonObj?.responseBody?.json?.['$'];
            if (!rawInner) return null;
            const inner = JSON.parse(rawInner);
            let codigo = null, referencia = null;

            if (Array.isArray(inner.fieldsMetadata)) {
                const fieldNames = inner.fieldsMetadata.map(f => f.fieldName);
                const codIdx = fieldNames.indexOf('CODPROD');
                const refIdx = fieldNames.indexOf('REFERENCIA');
                const rows = inner.data || inner.rows || inner.values || inner.records;
                if (Array.isArray(rows) && rows.length > 0) {
                    const row = rows[0];
                    if (Array.isArray(row)) {
                        if (codIdx !== -1 && row[codIdx]) codigo = String(row[codIdx]);
                        if (refIdx !== -1 && row[refIdx]) referencia = String(row[refIdx]);
                    } else if (typeof row === 'object') {
                        if (row.CODPROD) codigo = String(row.CODPROD);
                        if (row.REFERENCIA) referencia = String(row.REFERENCIA);
                    }
                }
            }

            if (!codigo) {
                const list = inner.suggestions || inner.records || inner.data || [];
                if (Array.isArray(list) && list.length > 0) {
                    const item = list[0];
                    if (item.CODPROD) codigo = String(item.CODPROD);
                    if (item.REFERENCIA) referencia = String(item.REFERENCIA);
                }
            }

            if (!codigo) {
                const m = rawInner.match(/"CODPROD"\s*:\s*"?(\d+)"?/i);
                if (m) codigo = m[1];
            }

            if (!referencia) {
                const m = rawInner.match(/"REFERENCIA"\s*:\s*"?([^",}]+)"?/i);
                if (m && m[1].trim() !== 'null') referencia = m[1].trim();
            }

            return codigo ? { codigo, referencia } : null;
        } catch (err) {
            return null;
        }
    }

    /**
     * Aplicação blindada de valores em inputs da Magis5
     * Suporta eventos nativos + protótipo + jQuery + updateExtraFields
     */
    function setInputValue(input, valor) {
        if (!input) return;
        input.focus();
        input.select?.();
        const setter = Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, 'value')?.set;
        if (setter) setter.call(input, valor); else input.value = valor;
        input.value = valor;

        // Disparo de eventos nativos
        input.dispatchEvent(new Event('input', { bubbles: true }));
        input.dispatchEvent(new Event('change', { bubbles: true }));
        input.dispatchEvent(new Event('focusout', { bubbles: true }));
        input.blur();

        // Suporte obrigatório ao jQuery da Magis5 (Metronic)
        try {
            const jq = window.jQuery || window.$;
            if (jq) {
                jq(input).val(valor).trigger('input').trigger('change').trigger('blur');
            }
        } catch (_) {}

        // Atualiza campos vinculados da Magis5
        const idMatch = (input.id || input.name || '').match(/(\d+)/);
        if (idMatch && typeof window.updateExtraFields === 'function') {
            try { window.updateExtraFields(parseInt(idMatch[1])); } catch (_) {}
        }
        highlight(input, '#10b981');
    }

        const entries = [];
        for (const inp of getSkuInputs()) {
            const sku = norm(inp.value); if (!sku) continue;
            const container = getRegistroContainer(inp);
            let temEan = false;
            for (const ei of (container ? container.querySelectorAll('input') : [])) {
                const id = (ei.id || ei.name || '').toLowerCase();
                if ((id.includes('ean') || id.includes('gtin')) && norm(ei.value)) { temEan = true; break; }
            }
            entries.push({ sku, precisaEan: !temEan });
        }
        return entries;
    }

            if (mgeSession) url += `&mgeSession=${mgeSession}`;

            const body = {
                serviceName: 'PesquisaSP.getSuggestion',
                requestBody: {
                    criteria: {
                        entityName: 'Produto',
                        compacted: false,
                        ignoreEntityCriteria: false,
                        limit: '5',
                        query: { '$': sku },
                        orderByDesc: false,
                        options: { showInactives: true },
                        externalCriteria: {
                            query: { '$': "((this.USOPROD <> 'S' AND this.USOPROD <> '3' ))" },
                            params: { param: [] }
                        }
                    },
                    clientEventList: {
                        clientEvent: [
                            { '$': 'br.com.sankhya.mge.info.metro.cubico' },
                            { '$': 'br.com.sankhya.actionbutton.clientconfirm' }
                        ]
                    }
                }
            };

            GM_xmlhttpRequest({
                method: 'POST',
                url: url,
                headers: {
                    'Accept': 'application/json, text/plain, */*',
                    'Content-Type': 'application/json; charset=UTF-8',
                    'sktk': sktk
                },
                data: JSON.stringify(body),
                timeout: 10000,
                onload: (res) => {
                    try {
                        if (res.status === 200) {
                            resolve(parseSankhyaResponse(JSON.parse(res.responseText)));
                        } else {
                            resolve(null);
                        }
                    } catch (_) { resolve(null); }
                },
                ontimeout: () => resolve(null),
                onerror: () => resolve(null)
            });
        });
    }

    function waitForSankhyaResults(timeoutMs = 60000) {
        return new Promise((resolve) => {
            const startTime = Date.now();
            const listenerId = GM_addValueChangeListener('brk_skus_resultado', (name, oldVal, newVal, remote) => {
                try {
                    const result = JSON.parse(newVal);
                    if (result && result.status === 'completo' && result.pares) {
                        GM_removeValueChangeListener(listenerId);
                        clearInterval(timeoutCheck);
                        log(`📡 Resultado recebido do Sankhya: ${result.pares.length} pares`, 'success');
                        resolve(result.pares);
                    } else if (result && result.status === 'erro') {
                        GM_removeValueChangeListener(listenerId);
                        clearInterval(timeoutCheck);
                        log(`❌ Sankhya erro: ${result.mensagem || '?'}`, 'warn');
                        resolve(null);
                    }
                } catch (e) {}
            });
            const timeoutCheck = setInterval(() => {
                if (Date.now() - startTime > timeoutMs) {
                    GM_removeValueChangeListener(listenerId);
                    clearInterval(timeoutCheck);
                    log(`⏰ Timeout aguardando resposta (${timeoutMs / 1000}s).`, 'warn');
                    resolve(null);
                }
            }, 3000);
            try {
                const existing = GM_getValue('brk_skus_resultado', '');
                if (existing) {
                    const result = JSON.parse(existing);
                    if (result && result.status === 'completo' && result.timestamp > startTime - 3000) {
                        GM_removeValueChangeListener(listenerId);
                        clearInterval(timeoutCheck);
                        resolve(result.pares);
                    }
                }
            } catch (_) {}
        });
    }

    /**
     * Motor Híbrido: Tenta API Direta na própria Magis5 (instantâneo!)
     * Se não tiver token, usa o canal Cross-Tab automaticamente.
     */
    async function obterCodigosSankhyaParaVariantes(skuEntries) {
        const sktk = GM_getValue(TOKEN_KEY, '');

        // 1. API DIRETA DA MAGIS5 (sem depender de outra aba aberta)
        if (sktk) {
            log(`⚡ Sessão Sankhya ativa! Consultando API direta na Magis5...`, 'info');
            showAction(`Consultando API Sankhya (${skuEntries.length} SKUs)...`, '⚡');

            const cache = (await GM_getValue(CACHE_KEY, null)) || {};
            const uniqueSkus = [...new Set(skuEntries.map(e => e.sku))];
            const resultMap = new Map();
            let index = 0;
            const CONCORRENCIA = 6;

            async function worker() {
                while (index < uniqueSkus.length) {
                    const i = index++;
                    const sku = uniqueSkus[i];
                    if (cache[sku]?.codigo) {
                        resultMap.set(sku, { codigo: cache[sku].codigo, referencia: cache[sku].referencia || '' });
                        log(`⚡ [Cache] ${sku} ➔ ${cache[sku].codigo}`);
                    } else {
                        try {
                            const res = await consultarSankhyaApiViaGm(sku);
                            if (res?.codigo) {
                                resultMap.set(sku, { codigo: res.codigo, referencia: res.referencia || '' });
                                cache[sku] = { codigo: res.codigo, referencia: res.referencia || '', t: Date.now() };
                                log(`✅ [API] ${sku} ➔ ${res.codigo}${res.referencia ? ' (Ref: ' + res.referencia + ')' : ''}`, 'success');
                            } else {
                                log(`⚠️ [API] ${sku}: não encontrado no Sankhya`, 'warn');
                            }
                        } catch (err) {
                            console.warn('[BRK Magis5] Erro API:', err);
                        }
                    }
                    await sleep(15);
                }
            }

            await Promise.all(Array.from({ length: Math.min(CONCORRENCIA, uniqueSkus.length) }, () => worker()));
            try { await GM_setValue(CACHE_KEY, cache); } catch(_) {}

            const pares = [];
            for (const entry of skuEntries) {
                const info = resultMap.get(entry.sku);
                if (info) {
                    pares.push({
                        sku: entry.sku,
                        codigo: info.codigo,
                        ean: (entry.precisaEan && info.referencia) ? info.referencia : ''
                    });
                }
            }

            if (pares.length > 0) {
                log(`✅ API Sankhya: ${pares.length}/${skuEntries.length} códigos obtidos com sucesso!`, 'success');
                return pares;
            }
            log(`⚠️ Nenhum código obtido via API direta. Tentando canal Cross-Tab...`, 'warn');
        } else {
            log(`ℹ️ Token da sessão Sankhya ainda não capturado. Abra o Sankhya em outra aba.`, 'warn');
        }

        // 2. FALLBACK CROSS-TAB (para aba do Sankhya aberta)
        showAction(`Aguardando Sankhya via Cross-Tab...`, '📡');
        GM_setValue('brk_skus_resultado', '');
        await sleep(300);
        sendSkusToSankhya(skuEntries);
        return await waitForSankhyaResults(60000);
    }

    function applySankhyaResults(pares) {
        if (!pares || pares.length === 0) return 0;
        const skuInputs = getSkuInputs();
        const inputMap = new Map();
        skuInputs.forEach(inp => inputMap.set(norm(inp.value).toUpperCase(), inp));

        let applied = 0;
        for (const par of pares) {
            const input = inputMap.get(par.sku.toUpperCase());
            if (!input) continue;

            // Troca o SKU da variante pelo Código Sankhya
            setInputValue(input, par.codigo);
            applied++;

            // Aplica EAN se disponível e campo estiver vazio
            if (par.ean) {
                const container = getRegistroContainer(input);
                if (container) {
                    for (const ei of container.querySelectorAll('input')) {
                        const id = (ei.id || ei.name || '').toLowerCase();
                        if ((id.includes('ean') || id.includes('gtin')) && !norm(ei.value)) {
                            setInputValue(ei, par.ean);
                            log(`  EAN aplicado: ${par.ean}`, 'info');
                            break;
                        }
                    }
                }
            }
        }
        return applied;
    }

    function parseInput(text) {
        const items = [];
        for (const line of text.split('\n').map(norm).filter(Boolean)) {
            const parts = line.split(/\t+/);
            const shopifyId = norm(parts[0]); if (!shopifyId) continue;
            const skuPairs = [];
            for (let i = 1; i < parts.length; i += 2) { const skuAtual = norm(parts[i]); const novoSku = norm(parts[i + 1] || ''); if (skuAtual && novoSku) skuPairs.push({ skuAtual, novoSku }); }
            items.push({ shopifyId, skuPairs });
        }
        return items;
    }

    function detectarTipoProduto(skuInputs) {
        for (const inp of skuInputs) { const sku = norm(inp.value).toUpperCase(); if (!sku) continue; if (sku.startsWith('ALL') || sku.startsWith('KIT')) return 'pular'; if (sku.includes('_FULL')) return 'revenda'; }
        return 'producao';
    }

    /**
     * Verifica se os SKUs já são códigos Sankhya válidos (numéricos curtos de 2 a 7 dígitos).
     * Evita falso positivo em EANs (8 a 14 dígitos) ou IDs longos de anúncio.
     */
    function skusJaTemCodigoSankhya(skuInputs) {
        let count = 0;
        for (const inp of skuInputs) {
            const sku = norm(inp.value);
            if (!sku) continue;
            if (/^\d{2,7}$/.test(sku)) count++;
            else return false;
        }
        return count > 0 && count === skuInputs.length;
    }

            status(`${label} ID: ${item.shopifyId}`);
            try {

                log(`${label} Tipo: ${tipoDetectado.toUpperCase()}${tipoOverride !== 'auto' ? ' → Override: ' + tipoFinal.toUpperCase() : ''}`);
                if (tipoFinal === 'pular') { log(`${label} ⏭ PULANDO (ALL/KIT)`, 'warn'); closeModals(); await sleep(1000); continue; }

                if (item.skuPairs.length > 0) {

                    const inputMap = new Map(); skuInputs.forEach(inp => inputMap.set(norm(inp.value).toUpperCase(), inp));
                    let applied = 0;
                    for (const pair of item.skuPairs) {
                        const input = inputMap.get(pair.skuAtual.toUpperCase());
                        if (input) { setInputValue(input, pair.novoSku); applied++; await sleep(250); }
                    }
                    log(`${label} ${applied}/${item.skuPairs.length} SKUs aplicados (manual).`);
                } else if (skusJaTemCodigoSankhya(skuInputs)) {

                        if (pares && pares.length > 0) {
                            const applied = applySankhyaResults(pares);
                            await sleep(cfg.delayStep);
                            log(`${label} ✅ ${applied}/${pares.length} códigos Sankhya aplicados automaticamente!`, 'success');
                        } else {
                            log(`${label} ⚠️ Sankhya não retornou códigos para estas variantes.`, 'warn');
                        }
                    }
                }

                if (tipoFinal === 'producao') {
                    let count = 0; for (const inp of skuInputs) { if (setVendidoDiversa(getRegistroContainer(inp), 'sim')) count++; } await sleep(cfg.delayStep);

    // ── MODO PÁGINA ──
    function scanPageSkus() {
        const skuInputs = getSkuInputs();
        const entries = [];
        for (const inp of skuInputs) {
            const sku = norm(inp.value); if (!sku) continue;
            const skuUpper = sku.toUpperCase();
            const container = getRegistroContainer(inp);
            let temEan = false;
            for (const ei of (container ? container.querySelectorAll('input') : [])) {
                const id = (ei.id || ei.name || '').toLowerCase();
                if ((id.includes('ean') || id.includes('gtin')) && norm(ei.value)) { temEan = true; break; }
            }
            let categoria;
            if (skuUpper.startsWith('ALL') || skuUpper.startsWith('KIT')) categoria = 'descartar';
            else if (skuUpper.includes('_FULL')) categoria = 'revenda';
            else if (/^\d{2,7}$/.test(sku)) categoria = 'ja_sankhya';
            else categoria = 'consultar';
            entries.push({ sku, input: inp, container, temEan, categoria });
        }
        return { total: entries.length, jaTemSankhya: entries.filter(e => e.categoria === 'ja_sankhya'), paraConsultar: entries.filter(e => e.categoria === 'consultar'), revenda: entries.filter(e => e.categoria === 'revenda'), descartar: entries.filter(e => e.categoria === 'descartar'), semEan: entries.filter(e => !e.temEan && e.categoria !== 'descartar'), all: entries };
    }

        }
        status('Configurando Diversa + Depósito...');

        status(onlyFirst ? '🧪 Teste concluído' : '✅ Concluído — Clique "Atualizar" para salvar!');

    async function checkPendingFromHookPrime() {
        try {
            const data = await new Promise((resolve) => {
                GM_xmlhttpRequest({
                    method: 'GET',
                    url: 'http://localhost:3000/api/pipeline/pending-ids',
                    timeout: 5000,
                    onload: (res) => {
                        try { resolve(JSON.parse(res.responseText)); }
                        catch (e) { resolve(null); }
                    },
                    onerror: () => resolve(null),
                    ontimeout: () => resolve(null),
                });
            });
            if (!data || !data.success || !data.ids || data.ids.length === 0) return;
            _hookPrimeSessionId = data.sessionId || '';
            const textarea = document.getElementById('brk-import-list');
            if (textarea && !textarea.value.trim()) {
                textarea.value = data.ids.join('\n');
                log(`📡 HOOK PRIME: ${data.ids.length} IDs carregados automaticamente (${data.store || 'unknown'})`, 'success');
            }
            // Auto-selecionar canal pela loja
            if (data.channel) {
                const channelSelect = document.getElementById('brk-import-channel');
                if (channelSelect) {
                    const opt = [...channelSelect.options].find(o => o.value === data.channel);
                    if (opt) {
                        channelSelect.value = data.channel;
                        opt.selected = true;
                        channelSelect.dispatchEvent(new Event('change', { bubbles: true }));
                        log(`🏪 Canal auto-selecionado: ${opt.textContent.trim()}`, 'success');
                    }
                }
            }
        } catch (e) {
            // HOOK PRIME offline — fluxo manual continua normalmente
        }
    }

        async function runSearch() {
            const rawLines = inputSkus.value.split('\n').map(norm); const skus = rawLines.filter(Boolean);
            if (!skus.length) { snkLog('Cole ao menos um código.', 'warn'); return; }
            const sktk = getActiveSktk();
            if (!sktk) { snkLog('⚠️ Token não capturado. Clique em um menu do Sankhya.', 'error'); return; }
            state.running = true; state.stopRequested = false;
            btnStart.style.display = 'none'; btnStop.style.display = 'inline-flex';
            statTotal.textContent = String(skus.length); statOk.textContent = '0'; statFail.textContent = '0'; statStatus.textContent = 'Buscando...';
            outputCod.value = 'Consultando...'; outputRef.value = 'Consultando...';
            const useCache = panel.querySelector('#snk-use-cache').checked;
            const cache = useCache ? await getSkuCache() : {};
            const uniqueSkus = unique(skus); const resultMap = new Map(); let concluidos = 0; const startTime = Date.now();
            snkLog(`🚀 Buscando ${skus.length} linhas (${uniqueSkus.length} únicos) — ${CONCORRENCIA} workers`, 'info');
            let index = 0;
            async function worker() {
                while (index < uniqueSkus.length) {
                    if (state.stopRequested) break;
                    const i = index++; const sku = uniqueSkus[i];
                    if (useCache && cache[sku]?.codigo) { resultMap.set(sku, { codigo: cache[sku].codigo, referencia: cache[sku].referencia || '', found: true }); snkLog(`⚡ [Cache] ${sku} ➔ ${cache[sku].codigo}`); }
                    else { try { const res = await fetchSkuDiretoApi(sku); if (res?.codigo) { resultMap.set(sku, { codigo: res.codigo, referencia: res.referencia || '', found: true }); if (useCache) await setSkuCache(sku, res.codigo, res.referencia); snkLog(`✅ ${sku} ➔ ${res.codigo}${res.referencia ? ` (Ref: ${res.referencia})` : ''}`, 'success'); } else { resultMap.set(sku, { codigo: '', referencia: '', found: false, motivo: 'Não encontrado' }); snkLog(`⚠️ ${sku}: Não encontrado`, 'warn'); } } catch (err) { resultMap.set(sku, { codigo: '', referencia: '', found: false, motivo: err.message }); snkLog(`⛔ ${sku}: ${err.message}`, 'error'); } }
                    concluidos++; statStatus.textContent = `${concluidos}/${uniqueSkus.length}`; await sleep(15);
                }
            }
            await Promise.all(Array.from({ length: Math.min(CONCORRENCIA, uniqueSkus.length) }, () => worker()));
            const codOut = [], refOut = []; let okCount = 0, failCount = 0; const fullResults = [];
            rawLines.forEach(sku => { if (!sku) { codOut.push(''); refOut.push(''); return; } const info = resultMap.get(sku); if (info?.found) { codOut.push(info.codigo); refOut.push(info.referencia); okCount++; fullResults.push({ sku, codigo: info.codigo, referencia: info.referencia, status: 'OK' }); } else { codOut.push(''); refOut.push(''); failCount++; fullResults.push({ sku, codigo: '', referencia: '', status: info?.motivo || 'Não encontrado' }); } });
            outputCod.value = codOut.join('\n'); outputRef.value = refOut.join('\n'); state.resultsList = fullResults;
            statOk.textContent = String(okCount); statFail.textContent = String(failCount);
            statStatus.textContent = state.stopRequested ? 'Interrompido' : 'Concluído';
            snkLog(`🏁 ${((Date.now() - startTime) / 1000).toFixed(1)}s! ✅ ${okCount} | ⚠ ${failCount}`, 'success');
            playBeep(); state.running = false; btnStart.style.display = 'inline-flex'; btnStop.style.display = 'none';
        }

        btnStart.onclick = runSearch;
        btnStop.onclick = () => { state.stopRequested = true; snkLog('Parada solicitada.', 'warn'); };
        panel.querySelector('#snk-btn-copy-table').onclick = () => { if (!state.resultsList.length) return; const h = ['SKU', 'CODPROD', 'REFERENCIA', 'STATUS'].join('\t'); const rows = state.resultsList.map(r => [r.sku, r.codigo, r.referencia, r.status].join('\t')); GM_setClipboard([h, ...rows].join('\n')); snkLog('📋 Tabela copiada!', 'success'); };

    // ── MODO AUTOMÁTICO (FALLBACK VIA LISTENER CROSS-TAB) ──
    async function processarListaAutomatica(skuEntries) {
        const sktk = getActiveSktk();
        if (!sktk) {
            GM_setValue('brk_skus_resultado', JSON.stringify({ timestamp: Date.now(), status: 'erro', mensagem: 'Token não capturado no Sankhya' }));
            return;
        }
        const cache = await getSkuCache();
        const pares = [];
        const uniqueSkus = unique(skuEntries.map(e => e.sku));
        const resultMap = new Map();
        let index = 0;

        async function worker() {
            while (index < uniqueSkus.length) {
                const i = index++;
                const sku = uniqueSkus[i];
                if (cache[sku]?.codigo) {
                    resultMap.set(sku, { codigo: cache[sku].codigo, referencia: cache[sku].referencia || '' });
                } else {
                    try {
                        const res = await fetchSkuDiretoApi(sku);
                        if (res?.codigo) {
                            resultMap.set(sku, { codigo: res.codigo, referencia: res.referencia || '' });
                            await setSkuCache(sku, res.codigo, res.referencia);
                        }
                    } catch (err) {}
                }
                await sleep(15);
            }
        }
        await Promise.all(Array.from({ length: Math.min(CONCORRENCIA, uniqueSkus.length) }, () => worker()));

        for (const entry of skuEntries) {
            const info = resultMap.get(entry.sku);
            if (info) pares.push({ sku: entry.sku, codigo: info.codigo, ean: (entry.precisaEan && info.referencia) ? info.referencia : '' });
        }

        GM_setValue('brk_skus_resultado', JSON.stringify({ timestamp: Date.now(), status: 'completo', pares, total: skuEntries.length, encontrados: pares.length }));
    }

    GM_addValueChangeListener('brk_skus_pendentes', (name, oldVal, newVal, remote) => {
        if (!newVal) return;
        try {
            const p = JSON.parse(newVal);
            if (p?.status === 'pendente' && p.skus?.length > 0) {
                processarListaAutomatica(p.skus);
            }
        } catch (_) {}
    });
